// --- 1. Expense Data Model ---
class Expense {
  final String id;
  final DateTime date;
  String category; // Mutable as per requirement 4
  final String details;
  final double amount;

  Expense({
    required this.id,
    required this.date,
    required this.category,
    required this.details,
    required this.amount,
  });

  // 12. toJson method
  Map<String, dynamic> toJson() {
    return {
      'id': id,
      'date': date.toIso8601String(),
      'category': category,
      'details': details,
      'amount': amount,
    };
  }

  // 12. fromJson method
  factory Expense.fromJson(Map<String, dynamic> json) {
    return Expense(
      id: json['id'],
      date: DateTime.parse(json['date']),
      category: json['category'],
      details: json['details'],
      amount: json['amount'],
    );
  }

  @override
  String toString() {
    return 'Expense(id: $id, date: $date, category: $category, details: $details, amount: $amount)';
  }
}

// Helper class for file management (13, 14, 15, 16)
class AppFileManager {
  static const String _fileName = 'expense_tracker_data.json';

  Future<File> _getLocalFile() async {
    final directory = await getApplicationDocumentsDirectory();
    return File('${directory.path}/$_fileName');
  }

  // 15. Reads data from local JSON file
  Future<Map<String, dynamic>> readData() async {
    try {
      final file = await _getLocalFile();
      if (await file.exists()) {
        final contents = await file.readAsString();
        final Map<String, dynamic> jsonMap = json.decode(contents);
        return jsonMap;
      }
    } catch (e) {
      debugPrint("Error reading data: $e");
    }
    // Default data if file doesn't exist or error occurs
    return {
      'expenses': [],
      'categories': [
        'Food',
        'Transport',
        'Entertainment',
        'Utilities',
        'Rent'
      ] // Initial default categories
    };
  }

  // 16. Writes data to local JSON file
  Future<void> writeData(List<Expense> expenses, List<String> categories) async {
    try {
      final file = await _getLocalFile();
      final Map<String, dynamic> data = {
        'expenses': expenses.map((e) => e.toJson()).toList(),
        'categories': categories,
      };
      final String jsonString = json.encode(data);
      await file.writeAsString(jsonString);
    } catch (e) {
      debugPrint("Error writing data: $e");
    }
  }
}

// --- Main App Widget ---
void main() {
  runApp(const MyApp());
}

class MyApp extends StatefulWidget {
  const MyApp({super.key});

  @override
  State<MyApp> createState() => _MyAppState();
}

class _MyAppState extends State<MyApp> {
  int _selectedIndex = 0; // For BottomNavigationBar
  final PageController _pageController = PageController();

  List<Expense> _expenses = [];
  List<String> _categories = [];
  bool _isLoading = true; // 15. Loading spinner while data initializes
  final AppFileManager _appFileManager = AppFileManager();
  final Uuid _uuid = const Uuid(); // For generating unique IDs

  // Dashboard state variables (7, 8, 9)
  String _currentFilter = 'Month'; // Default filter
  DateTime _currentDashboardPeriodStart = DateTime.now();

  @override
  void initState() {
    super.initState();
    _initData();
  }

  // 15. Reads initial data from file
  Future<void> _initData() async {
    final data = await _appFileManager.readData();
    setState(() {
      _expenses = (data['expenses'] as List).map((e) => Expense.fromJson(e)).toList();
      _categories = (data['categories'] as List).cast<String>();
      // Ensure there's always at least one category, even if loaded empty
      if (_categories.isEmpty) {
        _categories = ['Miscellaneous'];
      }
      _isLoading = false;
    });
  }

  // 16. Saves current state of expenses and categories to file
  Future<void> _saveData() async {
    await _appFileManager.writeData(_expenses, _categories);
  }

  // --- Crucial Logic 4: Update categories in expenses ---
  void _updateExpenseCategories(String oldCategory, String newCategory) {
    setState(() {
      for (var expense in _expenses) {
        if (expense.category == oldCategory) {
          expense.category = newCategory;
        }
      }
      _saveData(); // Save changes after updating expenses
    });
  }

  void _onItemTapped(int index) {
    setState(() {
      _selectedIndex = index;
      _pageController.jumpToPage(index);
    });
  }

  // --- Dashboard Filtering Logic ---
  List<Expense> _getFilteredExpensesForDashboard() {
    DateTime periodStart = _currentDashboardPeriodStart;
    DateTime periodEnd;

    // Normalize periodStart to the actual beginning of the selected period type
    DateTime normalizedStart;

    if (_currentFilter == 'Week') {
      // Find the Monday of the current week (or Sunday based on locale)
      // Dart's weekday starts with Monday=1, Sunday=7
      normalizedStart = periodStart.subtract(Duration(days: periodStart.weekday - 1));
      normalizedStart = DateTime(normalizedStart.year, normalizedStart.month, normalizedStart.day);
      periodEnd = normalizedStart.add(const Duration(days: 7)).subtract(const Duration(seconds: 1));
    } else if (_currentFilter == 'Month') {
      normalizedStart = DateTime(periodStart.year, periodStart.month, 1);
      periodEnd = DateTime(periodStart.year, periodStart.month + 1, 0) // Last day of current month
          .add(const Duration(days: 1)) // Add 1 day to get to the next month's first day
          .subtract(const Duration(seconds: 1)); // Subtract 1 second to get last second of current month
    } else {
      // 'Year'
      normalizedStart = DateTime(periodStart.year, 1, 1);
      periodEnd = DateTime(periodStart.year + 1, 1, 1).subtract(const Duration(seconds: 1));
    }

    return _expenses.where((expense) {
      // Ensure expense date is strictly after (or equal to start of day) and strictly before (end of day)
      return expense.date.isAfter(normalizedStart.subtract(const Duration(seconds: 1))) &&
             expense.date.isBefore(periodEnd.add(const Duration(seconds: 1)));
    }).toList();
  }

  // 8. Navigates backward/forward in time based on filter
  void _navigatePeriod(int direction) {
    // direction: -1 for backward, 1 for forward
    setState(() {
      if (_currentFilter == 'Week') {
        _currentDashboardPeriodStart =
            _currentDashboardPeriodStart.add(Duration(days: 7 * direction));
      } else if (_currentFilter == 'Month') {
        _currentDashboardPeriodStart =
            DateTime(_currentDashboardPeriodStart.year, _currentDashboardPeriodStart.month + direction, 1);
      } else {
        // Year
        _currentDashboardPeriodStart =
            DateTime(_currentDashboardPeriodStart.year + direction, 1, 1);
      }
    });
  }

  // 9. Displays the currently viewed period
  String _getPeriodDisplayText() {
    DateTime start = _currentDashboardPeriodStart;
    if (_currentFilter == 'Week') {
      DateTime weekStart = start.subtract(Duration(days: start.weekday - 1)); // Monday
      DateTime weekEnd = weekStart.add(const Duration(days: 6)); // Sunday
      return '${DateFormat('MMM dd').format(weekStart)} - ${DateFormat('MMM dd, yyyy').format(weekEnd)}';
    } else if (_currentFilter == 'Month') {
      return DateFormat('MMMM yyyy').format(start);
    } else {
      // Year
      return DateFormat('yyyy').format(start);
    }
  }

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Expense Tracker',
      theme: ThemeData(
        primarySwatch: Colors.blue,
        useMaterial3: true,
      ),
      home: _isLoading
          ? const Scaffold(
              body: Center(
                child: CircularProgressIndicator(), // 15. Loading spinner
              ),
            )
          : Scaffold(
              appBar: AppBar(
                title: const Text('Expense Tracker'),
              ),
              body: PageView(
                controller: _pageController,
                onPageChanged: (index) {
                  setState(() {
                    _selectedIndex = index;
                  });
                },
                children: [
                  // Dashboard Tab
                  DashboardScreen(
                    expenses: _expenses,
                    categories: _categories, // Pass categories for potential future use
                    currentFilter: _currentFilter,
                    onFilterChanged: (filter) {
                      setState(() {
                        _currentFilter = filter;
                        // Reset period start to today's equivalent when filter changes
                        _currentDashboardPeriodStart = DateTime.now();
                      });
                    },
                    currentDashboardPeriodStart: _currentDashboardPeriodStart,
                    getFilteredExpensesForDashboard: _getFilteredExpensesForDashboard,
                    getPeriodDisplayText: _getPeriodDisplayText,
                    navigatePeriod: _navigatePeriod,
                  ),
                  // Add Expense Tab
                  AddExpenseScreen(
                    categories: _categories,
                    uuidGenerator: _uuid,
                    onAddExpense: (expense) {
                      setState(() {
                        _expenses.add(expense);
                        _saveData(); // 16. Save on add expense
                      });
                      _pageController.jumpToPage(0); // Auto-switch to Dashboard
                    },
                  ),
                  // Categories Tab
                  CategoriesScreen(
                    categories: _categories,
                    onAddCategory: (category) {
                      setState(() {
                        if (!_categories.contains(category)) {
                          _categories.add(category);
                          _saveData(); // 16. Save on add category
                        } else {
                          // Optionally show a snackbar if category already exists
                          ScaffoldMessenger.of(context).showSnackBar(
                            SnackBar(content: Text('Category "$category" already exists!')),
                          );
                        }
                      });
                    },
                    onDeleteCategory: (category) {
                      setState(() {
                        _categories.remove(category);
                        // Reassign expenses from deleted category to a 'Miscellaneous'
                        for (var expense in _expenses) {
                          if (expense.category == category) {
                            expense.category = 'Miscellaneous'; // Reassign to a default category
                          }
                        }
                        if (!_categories.contains('Miscellaneous') && _expenses.any((e) => e.category == 'Miscellaneous')) {
                           _categories.add('Miscellaneous'); // Add 'Miscellaneous' if it's not there and needed
                        }
                        _saveData(); // 16. Save on delete category
                      });
                    },
                    onEditCategory: (oldCategory, newCategory) {
                      if (oldCategory == newCategory) return; // No change
                      setState(() {
                        int index = _categories.indexOf(oldCategory);
                        if (index != -1) {
                          _categories[index] = newCategory;
                          // 4. Crucial Logic: Update all existing expenses with new category name
                          _updateExpenseCategories(oldCategory, newCategory);
                          _saveData(); // 16. Save on edit category
                        }
                      });
                    },
                  ),
                ],
              ),
              // 2. BottomNavigationBar
              bottomNavigationBar: BottomNavigationBar(
                items: const <BottomNavigationBarItem>[
                  BottomNavigationBarItem(
                    icon: Icon(Icons.dashboard),
                    label: 'Dashboard',
                  ),
                  BottomNavigationBarItem(
                    icon: Icon(Icons.add_circle),
                    label: 'Add Expense',
                  ),
                  BottomNavigationBarItem(
                    icon: Icon(Icons.category),
                    label: 'Categories',
                  ),
                ],
                currentIndex: _selectedIndex,
                selectedItemColor: Theme.of(context).primaryColor,
                onTap: _onItemTapped,
              ),
            ),
    );
  }
}

// --- Tab 1: Dashboard Screen ---
class DashboardScreen extends StatelessWidget {
  final List<Expense> expenses; // All expenses (passed for context)
  final List<String> categories; // All categories (passed for context)
  final String currentFilter; // 'Week', 'Month', 'Year'
  final ValueChanged<String> onFilterChanged; // Callback to change filter
  final DateTime currentDashboardPeriodStart; // Start of current period
  final List<Expense> Function() getFilteredExpensesForDashboard; // Function to get filtered expenses
  final String Function() getPeriodDisplayText; // Function to get period display text
  final ValueChanged<int> navigatePeriod; // Callback to navigate period (-1 or 1)

  const DashboardScreen({
    super.key,
    required this.expenses,
    required this.categories,
    required this.currentFilter,
    required this.onFilterChanged,
    required this.currentDashboardPeriodStart,
    required this.getFilteredExpensesForDashboard,
    required this.getPeriodDisplayText,
    required this.navigatePeriod,
  });

  @override
  Widget build(BuildContext context) {
    final List<Expense> filteredExpenses = getFilteredExpensesForDashboard();

    // 10. Calculate Total Spend
    final double totalSpend = filteredExpenses.fold(0.0, (sum, item) => sum + item.amount);

    // 11. Group expenses by category for PieGraph
    final Map<String, double> categoryTotals = {};
    for (var expense in filteredExpenses) {
      categoryTotals.update(expense.category, (value) => value + expense.amount,
          ifAbsent: () => expense.amount);
    }

    // A list of distinct colors for pie chart sections
    final List<Color> pieColors = [
      Colors.blue, Colors.green, Colors.red, Colors.purple,
      Colors.orange, Colors.teal, Colors.pink, Colors.brown,
      Colors.indigo, Colors.amber, Colors.cyan, Colors.deepOrange,
      Colors.lime, Colors.deepPurple,
    ];

    return Padding(
      padding: const EdgeInsets.all(16.0),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          // 7. SegmentedButton for filters
          SegmentedButton<String>(
            segments: const <ButtonSegment<String>>[
              ButtonSegment<String>(
                  value: 'Week', label: Text('Week'), icon: Icon(Icons.calendar_view_week)),
              ButtonSegment<String>(
                  value: 'Month', label: Text('Month'), icon: Icon(Icons.calendar_view_month)),
              ButtonSegment<String>(
                  value: 'Year', label: Text('Year'), icon: Icon(Icons.calendar_today)),
            ],
            selected: <String>{currentFilter},
            onSelectionChanged: (Set<String> newSelection) {
              if (newSelection.isNotEmpty) {
                onFilterChanged(newSelection.first);
              }
            },
          ),
          const SizedBox(height: 16),
          // 8, 9. Navigation arrows and period display
          Row(
            mainAxisAlignment: MainAxisAlignment.spaceBetween,
            children: [
              IconButton(
                icon: const Icon(Icons.arrow_left),
                onPressed: () => navigatePeriod(-1),
              ),
              Expanded(
                child: Text(
                  getPeriodDisplayText(),
                  textAlign: TextAlign.center,
                  style: Theme.of(context).textTheme.titleLarge,
                ),
              ),
              IconButton(
                icon: const Icon(Icons.arrow_right),
                onPressed: () => navigatePeriod(1),
              ),
            ],
          ),
          const SizedBox(height: 16),
          // 10. Total Spend
          Card(
            elevation: 2,
            child: Padding(
              padding: const EdgeInsets.all(16.0),
              child: Column(
                children: [
                  Text('Total Spend:', style: Theme.of(context).textTheme.titleMedium),
                  Text(
                    NumberFormat.currency(symbol: 'Rs.', decimalDigits: 2).format(totalSpend),
                    style: Theme.of(context).textTheme.headlineMedium?.copyWith(
                          fontWeight: FontWeight.bold,
                          color: Colors.red,
                        ),
                  ),
                ],
              ),
            ),
          ),
          const SizedBox(height: 16),
          // 11. PieGraph with total spent per category
          Expanded(
            child: Card(
              elevation: 2,
              child: Padding(
                padding: const EdgeInsets.all(16.0),
                child: categoryTotals.isEmpty
                    ? Center(
                        child: Text(
                          'No expenses for this period. Add some!',
                          style: Theme.of(context).textTheme.titleMedium,
                          textAlign: TextAlign.center,
                        ),
                      )
                    : PieChart(
                        PieChartData(
                          sections: categoryTotals.entries.map((entry) {
                            final double percentage = totalSpend > 0 ? (entry.value / totalSpend) * 100 : 0;
                            final int colorIndex = categoryTotals.keys.toList().indexOf(entry.key) % pieColors.length;
                            return PieChartSectionData(
                              color: pieColors[colorIndex],
                              value: entry.value,
                              title:
                                  '${entry.key}\n${NumberFormat.currency(symbol: '\$', decimalDigits: 0).format(entry.value)} (${percentage.toStringAsFixed(0)}%)',
                              radius: 80,
                              titleStyle: const TextStyle(
                                  fontSize: 12,
                                  fontWeight: FontWeight.bold,
                                  color: Colors.white),
                              titlePositionPercentageOffset: 0.55,
                            );
                          }).toList(),
                          sectionsSpace: 2,
                          centerSpaceRadius: 40,
                          startDegreeOffset: -90,
                          borderData: FlBorderData(show: false),
                        ),
                      ),
              ),
            ),
          ),
        ],
      ),
    );
  }
}

// --- Tab 2: Add Expense Screen ---
class AddExpenseScreen extends StatefulWidget {
  final List<String> categories;
  final ValueChanged<Expense> onAddExpense;
  final Uuid uuidGenerator; // Pass Uuid instance for consistent ID generation

  const AddExpenseScreen({
    super.key,
    required this.categories,
    required this.onAddExpense,
    required this.uuidGenerator,
  });

  @override
  State<AddExpenseScreen> createState() => _AddExpenseScreenState();
}

class _AddExpenseScreenState extends State<AddExpenseScreen> {
  final _formKey = GlobalKey<FormState>();
  final TextEditingController _amountController = TextEditingController();
  final TextEditingController _detailsController = TextEditingController();
  String? _selectedCategory;
  DateTime _selectedDate = DateTime.now(); // 6. Date picker defaults to today

  @override
  void initState() {
    super.initState();
    // 5. Initialize selected category if categories are available
    if (widget.categories.isNotEmpty) {
      _selectedCategory = widget.categories.first;
    }
  }

  // Helper to check if _selectedCategory is still in the current categories list
  bool _isValidCategory(String? category) {
    return category != null && widget.categories.contains(category);
  }

  Future<void> _selectDate(BuildContext context) async {
    final DateTime? picked = await showDatePicker(
      context: context,
      initialDate: _selectedDate,
      firstDate: DateTime(2000),
      lastDate: DateTime(2101),
    );
    if (picked != null && picked != _selectedDate) {
      setState(() {
        _selectedDate = picked;
      });
    }
  }

  void _submitForm() {
    if (_formKey.currentState!.validate() && _selectedCategory != null) {
      final newExpense = Expense(
        id: widget.uuidGenerator.v4(), // Generate unique ID
        date: _selectedDate,
        category: _selectedCategory!,
        details: _detailsController.text,
        amount: double.parse(_amountController.text),
      );
      widget.onAddExpense(newExpense);

      // Clear form fields
      _amountController.clear();
      _detailsController.clear();
      setState(() {
        _selectedDate = DateTime.now();
        // Reset category dropdown, ensuring it's still a valid category
        _selectedCategory = widget.categories.isNotEmpty ? widget.categories.first : null;
      });
    }
  }

  @override
  void didUpdateWidget(covariant AddExpenseScreen oldWidget) {
    super.didUpdateWidget(oldWidget);
    // 5. Crucial Logic: Dynamically update dropdown if categories change
    // If selected category is no longer in the list, or no category is selected
    if (!_isValidCategory(_selectedCategory) && widget.categories.isNotEmpty) {
      setState(() {
        _selectedCategory = widget.categories.first; // Set to the first available category
      });
    } else if (widget.categories.isEmpty) {
      setState(() {
        _selectedCategory = null; // No categories available
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return SingleChildScrollView(
      padding: const EdgeInsets.all(16.0),
      child: Form(
        key: _formKey,
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            TextFormField(
              controller: _amountController,
              decoration: const InputDecoration(
                labelText: 'Amount (Rs.)',
                border: OutlineInputBorder(),
              ),
              keyboardType: TextInputType.number,
              validator: (value) {
                if (value == null || value.isEmpty) {
                  return 'Please enter an amount';
                }
                if (double.tryParse(value) == null) {
                  return 'Please enter a valid number';
                }
                if (double.parse(value) <= 0) {
                  return 'Amount must be positive';
                }
                return null;
              },
            ),
            const SizedBox(height: 16),
            TextFormField(
              controller: _detailsController,
              decoration: const InputDecoration(
                labelText: 'Details (e.g., Coffee at Starbucks)',
                border: OutlineInputBorder(),
              ),
              validator: (value) {
                if (value == null || value.isEmpty) {
                  return 'Please enter details';
                }
                return null;
              },
            ),
            const SizedBox(height: 16),
            // 5. Category Dropdown
            DropdownButtonFormField<String>(
              value: _selectedCategory,
              decoration: const InputDecoration(
                labelText: 'Category',
                border: OutlineInputBorder(),
              ),
              items: widget.categories.map((category) {
                return DropdownMenuItem(
                  value: category,
                  child: Text(category),
                );
              }).toList(),
              onChanged: (String? newValue) {
                setState(() {
                  _selectedCategory = newValue;
                });
              },
              validator: (value) {
                if (value == null || value.isEmpty) {
                  return 'Please select a category';
                }
                return null;
              },
              // If no categories are available, show a hint
              hint: widget.categories.isEmpty
                  ? const Text('No categories available. Add some in the Categories tab.')
                  : const Text('Select a category'),
              isExpanded: true,
            ),
            const SizedBox(height: 16),
            // 6. Date Picker
            ListTile(
              title: const Text('Date'),
              subtitle: Text(DateFormat('yyyy-MM-dd').format(_selectedDate)),
              trailing: const Icon(Icons.calendar_today),
              onTap: () => _selectDate(context),
              shape: OutlineInputBorder(borderRadius: BorderRadius.circular(4)),
            ),
            const SizedBox(height: 24),
            ElevatedButton(
              onPressed: _selectedCategory != null && widget.categories.isNotEmpty ? _submitForm : null,
              style: ElevatedButton.styleFrom(
                padding: const EdgeInsets.symmetric(vertical: 16),
                textStyle: const TextStyle(fontSize: 18),
              ),
              child: const Text('Add Expense'),
            ),
            if (widget.categories.isEmpty) // Show a hint if no categories exist
              Padding(
                padding: const EdgeInsets.only(top: 8.0),
                child: Text(
                  'Please add categories in the "Categories" tab first.',
                  style: TextStyle(color: Theme.of(context).colorScheme.error),
                  textAlign: TextAlign.center,
                ),
              ),
          ],
        ),
      ),
    );
  }
}

// --- Tab 3: Categories Screen ---
class CategoriesScreen extends StatefulWidget {
  final List<String> categories;
  final ValueChanged<String> onAddCategory;
  final ValueChanged<String> onDeleteCategory;
  final Function(String oldCategory, String newCategory) onEditCategory;

  const CategoriesScreen({
    super.key,
    required this.categories,
    required this.onAddCategory,
    required this.onDeleteCategory,
    required this.onEditCategory,
  });

  @override
  State<CategoriesScreen> createState() => _CategoriesScreenState();
}

class _CategoriesScreenState extends State<CategoriesScreen> {
  final TextEditingController _newCategoryController = TextEditingController();

  // 3. Edit a category name via a dialog
  void _showEditCategoryDialog(BuildContext context, String currentCategory) {
    final TextEditingController editController = TextEditingController(text: currentCategory);
    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('Edit Category'),
        content: TextField(
          controller: editController,
          decoration: const InputDecoration(labelText: 'Category Name'),
          autofocus: true,
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(ctx).pop(),
            child: const Text('Cancel'),
          ),
          ElevatedButton(
            onPressed: () {
              final newName = editController.text.trim();
              if (newName.isNotEmpty) {
                if (newName != currentCategory) {
                  if (!widget.categories.contains(newName)) { // Prevent duplicate category names
                    widget.onEditCategory(currentCategory, newName);
                    Navigator.of(ctx).pop();
                  } else {
                    ScaffoldMessenger.of(ctx).showSnackBar(
                      SnackBar(content: Text('Category "$newName" already exists!')),
                    );
                  }
                } else {
                  Navigator.of(ctx).pop(); // No change, just close
                }
              } else {
                 ScaffoldMessenger.of(ctx).showSnackBar(
                    const SnackBar(content: Text('Category name cannot be empty!')),
                  );
              }
            },
            child: const Text('Save'),
          ),
        ],
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.all(16.0),
      child: Column(
        children: [
          // 3. Add a new category via a text field
          Row(
            children: [
              Expanded(
                child: TextField(
                  controller: _newCategoryController,
                  decoration: const InputDecoration(
                    labelText: 'New Category Name',
                    border: OutlineInputBorder(),
                  ),
                  onSubmitted: (_) { // Allows adding by pressing Enter key
                    if (_newCategoryController.text.trim().isNotEmpty) {
                      widget.onAddCategory(_newCategoryController.text.trim());
                      _newCategoryController.clear();
                    }
                  },
                ),
              ),
              const SizedBox(width: 8),
              ElevatedButton(
                onPressed: () {
                  if (_newCategoryController.text.trim().isNotEmpty) {
                    widget.onAddCategory(_newCategoryController.text.trim());
                    _newCategoryController.clear();
                  }
                },
                style: ElevatedButton.styleFrom(
                  padding: const EdgeInsets.symmetric(vertical: 16),
                ),
                child: const Text('Add'),
              ),
            ],
          ),
          const SizedBox(height: 16),
          // List of categories with edit/delete options
          Expanded(
            child: widget.categories.isEmpty
                ? const Center(child: Text('No categories added yet.'))
                : ListView.builder(
                    itemCount: widget.categories.length,
                    itemBuilder: (context, index) {
                      final category = widget.categories[index];
                      return Card(
                        margin: const EdgeInsets.symmetric(vertical: 4),
                        child: ListTile(
                          title: Text(category),
                          trailing: Row(
                            mainAxisSize: MainAxisSize.min,
                            children: [
                              // 3. Edit icon
                              IconButton(
                                icon: const Icon(Icons.edit, color: Colors.blue),
                                onPressed: () => _showEditCategoryDialog(context, category),
                              ),
                              // 3. Delete icon
                              IconButton(
                                icon: const Icon(Icons.delete, color: Colors.red),
                                onPressed: () {
                                  // Add confirmation dialog for deletion
                                  showDialog(
                                    context: context,
                                    builder: (ctx) => AlertDialog(
                                      title: const Text('Delete Category'),
                                      content: Text(
                                          'Are you sure you want to delete "$category"? Expenses currently assigned to this category will be re-assigned to "Miscellaneous".'),
                                      actions: [
                                        TextButton(
                                          onPressed: () => Navigator.of(ctx).pop(),
                                          child: const Text('Cancel'),
                                        ),
                                        ElevatedButton(
                                          onPressed: () {
                                            widget.onDeleteCategory(category);
                                            Navigator.of(ctx).pop();
                                          },
                                          style: ElevatedButton.styleFrom(
                                              backgroundColor: Colors.red),
                                          child: const Text('Delete',
                                              style: TextStyle(color: Colors.white)),
                                        ),
                                      ],
                                    ),
                                  );
                                },
                              ),
                            ],
                          ),
                        ),
                      );
                    },
                  ),
          ),
        ],
      ),
    );
  }
